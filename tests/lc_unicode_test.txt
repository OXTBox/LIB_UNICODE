  stop using stack "lib_unicode"
  delete stack "lib_unicode"
end if
put "//wsl$/Ubuntu/home/igame3d/projects/LIB_UNICODE/" into tDir
start using stack (tDir & "livecode/lib_unicode.livecodescript")
put empty into tLog
put "folder: " & (Unicode.Folder() & "/" is tDir or Unicode.Folder() & "/" is "/home/igame3d/projects/LIB_UNICODE/") & return after tLog
put Unicode.Char("house") into tHouse
put "house char: " & (Unicode.Code(tHouse) is "U+2302") & return after tLog
put "gear = tools menu: " & (Unicode.Code(Unicode.Char("tools menu")) is "U+2699") & return after tLog
put "pencil = editor: " & (Unicode.Meaning(numToCodepoint(baseConvert("270E", 16, 10))) is "editor") & return after tLog
put "procedure: " & (Unicode.Procedure(tHouse) is "Proc.House") & return after tLog
put "astral code (car): " & (Unicode.Code(Unicode.Char("car")) is "U+1F697") & return after tLog

put "entities: " & (the number of lines of Unicode.List("entity") >= 6) & return after tLog
put "registry self-check empty: " & (Unicode.Check() is empty) & return after tLog

end repeat

put "cloud 5 balls of 32 faces: " & (the number of lines of tCl["faces"] is 160) & return after tLog
put "cloud floats (bottom pole of ball 1 at -60): " & (abs(tCl["vy"][26] + 60) < 0.01) && tCl["vy"][26] & return after tLog
put Unicode.Build(Unicode.Char("storm cloud"), 40) into tS
put "storm cloud: cloud + 6 bolt points, 5 lines: " & (tS["count"] is 136 and the number of lines of tS["lines"] is 5) & return after tLog
put "tool, no procedure: " & (Unicode.Build(Unicode.Char("tools menu")) is empty and Unicode.LastError() contains "no procedure") & return after tLog
return tLog
